  (defun mycaml-match-string (num &optional string)

    "Return string of text matched by last search, without properties.
  NUM specifies which parenthesized expression in the last regexp.
  Value is nil if NUMth pair didn't match, or there were less than NUM
  pairs.  Zero means the entire text matched by the whole regexp or
  whole string."

    (let* ((data (match-data))
           (begin (nth (* 2 num) data))
           (end (nth (1+ (* 2 num)) data)))
      (if string (substring string begin end)
        (buffer-substring-no-properties begin end))))

  (defun c-find-alternate-file ()
    (interactive)
    (let ((name (buffer-file-name)))
      (if (string-match "^\\(.*\\)\\.\\(h\\|c\\)$" name)
          (find-file
           (concat
            (mycaml-match-string 1 name)
            (if (string= "c" (mycaml-match-string 2 name)) ".h" ".c"))))))

  (defun cpp-find-alternate-file ()
    (interactive)
    (let ((name (buffer-file-name)))
      (if (string-match "^\\(.*\\)\\.\\(h\\|cpp\\)$" name)
          (find-file
           (concat
            (mycaml-match-string 1 name)
            (if (string= "cpp" (mycaml-match-string 2 name)) ".h" ".cpp"))))))

#+BEGIN_SRC emacs-lisp
  ;; Control whether to add ;; at the end of the declaration
  ;; TODO: look in tuareg whether there is a similar thing
  (defvar rb/ocaml/commacomma-ends-sentence t)

  (defun rb/ocaml-pp-argtypes (argtypes )
    "docstring"
    (let ((list argtypes)
          (param-stack)
          (current-param)
          (params))
      (while list
        (let ((ocamldoc-keyword
               (if (consp (cdr list)) ;; the last type is the return type
                   "@param"
                 "@return"
                 )
               )
              (param (car list)))
          (cond ((string-equal
                  (substring param 0 1)
                  "(")
                 (progn
                   ;; this starts a function type
                   (push
                    current-param
                    param-stack)
                   (push ocamldoc-keyword param-stack)
                   (setq current-param
                         (substring param 1))))
                ((string-equal (substring param -1) ")")
                 (progn
                   (push (concat (pop param-stack) " " current-param " -> " (substring param 0 -1)) params)
                   (setq current-param (pop param-stack))
                   ))
                (param-stack
                 ;; we're handling a function
                 (setq current-param (concat current-param " -> " param))
                 )
                (t (push (format "%s %s" ocamldoc-keyword param) params))
                )
          (setq list (cdr list)))
        )
      (string-join (reverse
                    params) "\n")
      )
    )

  (rb/ocaml-pp-argtypes '("foo" "bar" "baz"))

  ;; require merlin / tuareg
  ;;
  ;; when pointing at type it produces
  ;; val type: type 'a printable = 'a * (Format.formatter -> 'a -> unit) ;;
  (defun rb/ocaml-export-at-point ()
    "Export the type signature of the symbol at point.
     In particular, if you want to export the type of f, where f for example is
     let f x = (* body *) ... your point should be right before f occurs.
    "
    (interactive)
    (when
        (let ((fname (buffer-file-name (current-buffer))))
          (and fname (equal "ml" (file-name-extension fname))))
      ;; check thing at point is not a keyword
      (message (format "OCaml looking at %s" (thing-at-point 'symbol)))
      ;; Assume we're looking at the symbole we want to export
      ;; An improvement here would be to search for the name of the enclosing function/module/type definition
      (let (ocaml-identifier (thing-at-point 'symbol))
        (tuareg-beginning-of-defun)
        (let ((kw (thing-at-point 'symbol))
              (msitfor merlin-allow-sit-for)
              ;; default dummy type which has at least a meaning in OCaml
              (ocaml-kw-type nil) ;; has an influence on the conditional handling below
              (ocaml-val-type "unit"))

          (cond ((equal kw "module")
                 (progn
                   (setq ocaml-kw-type 'module)
                   (forward-word)
                   (forward-char)
                   (setq ocaml-identifier
                         (thing-at-point 'symbol))))
               
                ((equal kw "let")
                 (progn
                   (setq ocaml-kw-type 'val)
                   (forward-word)
                   (forward-char)
                   (setq ocaml-identifier
                         (thing-at-point 'symbol))))
               
                ((equal kw "type")
                 (setq ocaml-identifier nil))

                (t (message "falled through")
                   (setq ocaml-identifier "not-yet-implemented-case")))
          (setq merlin-allow-sit-for nil)
          (merlin-type-enclosing) ;; Seems to block until next keystroke
          (next-line)
          (beginning-of-line)
          (with-current-buffer merlin-type-buffer-name
            (setq ocaml-val-type (buffer-string))) ;
          ;; Assume current buffer ends with .ml
          (tuareg-find-alternate-file)
          ;; insert the signature at the end of the mli file
          (end-of-buffer)
          (newline)
          (message "kw: %s, id: %s" ocaml-kw-type ocaml-identifier)
          (insert
           (let ((exported
                  (if (and ocaml-kw-type ocaml-identifier)
                      (let* ((arg-types (split-string
                                         ocaml-val-type
                                         "->" t split-string-default-separators)))
                        (message "%s" arg-types)
                        (format
                         "(** [%s] \n%n*)\n%s %s: %s"
                         ocaml-identifier
                         (rb/ocaml-pp-argtypes
                          arg-types)
                         ocaml-kw-type
                         ocaml-identifier
                         ocaml-val-type)) 
                    ;; We were looking at like type, that is reprinted entirely
                    ;; in merlin-buffer
                    ocaml-val-type)))
             (format "\n%s%s"
                     exported
                     (if rb/ocaml/commacomma-ends-sentence " ;;" ""))))
          ;; reset merlin-allow-sit-for to whatever value it had before
          (setq merlin-allow-sit-for msitfor)))))

* Finding a project root

Try =project= then =projectile=. This should be smart enough since =project= is
in Emacs default distribution.

#+begin_src emacs-lisp
(defun rb--find-project-root ()
  "Find the project root directory.
Tries project.el first, then projectile, then falls back to `default-directory'."
  (or (and (fboundp 'project-root)
           (when-let ((proj (project-current)))
             (project-root proj)))
      (and (fboundp 'projectile-project-root)
           (projectile-project-root))
      default-directory))
#+end_src

** Project scratch buffers

=C-c s= creates (or visits) a scratch buffer named after the current project.
Falls back to =rb--find-project-root= which already tries project.el →
projectile → default-directory.

#+begin_src emacs-lisp
(defun rb/project-name ()
  "Return a short name for the current project, or \"scratch\"."
  (let ((root (rb--find-project-root)))
    (if (string= root default-directory)
        "scratch"
      (file-name-nondirectory (directory-file-name root)))))

(defun rb/new-project-scratch ()
  "Create or switch to a scratch buffer for the current project.
The buffer is named *scratch[PROJECT]* and uses fundamental-mode."
  (interactive)
  (let* ((name (rb/project-name))
         (buf  (get-buffer-create (format "*scratch[%s]*" name))))
    (switch-to-buffer buf)
    (when (zerop (buffer-size))
      (fundamental-mode)
      (insert (format ";; scratch — %s\n\n" name)))
    (setq buffer-read-only nil)))

(global-set-key (kbd "C-c s") #'rb/new-project-scratch)
#+end_src

* Column reading with follow-mode

Spread one buffer across several side-by-side windows and let =follow-mode=
scroll them as a single continuous column — useful to read a long file without
constant scrolling.

#+begin_src emacs-lisp
(defun column-read-buffer (buffer &optional n)
  "Read BUFFER across several side-by-side windows with `follow-mode'.
Delete all other windows, split the selected window into N extra
side-by-side windows (N defaults to 2), point every window at BUFFER,
and enable Follow mode so they scroll as one continuous column of text."
  (interactive "bBuffer: ")
  (let ((buffer (get-buffer buffer))
        (n (or n 2)))
    (switch-to-buffer buffer)
    (delete-other-windows)
    (dotimes (_ n)
      (split-window-right))
    (balance-windows)
    (dolist (win (window-list nil 'no-minibuffer))
      (set-window-buffer win buffer))
    (with-current-buffer buffer
      (follow-mode 1))))

(defun column-read-current-buffer (&optional n)
  "Run `column-read-buffer' on the current buffer.
With a prefix argument, add N extra windows instead of the default 2."
  (interactive "P")
  (column-read-buffer (current-buffer)
                      (and n (prefix-numeric-value n))))
#+end_src
