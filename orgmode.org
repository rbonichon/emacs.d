* Study sessions

Plan-aware helpers for study plans organised as =INDEX.org= +
=week-*.org= sibling files. Canonical layout under
=~/documents/org/study-plans/=, with each plan as its own
subdirectory. Each =* Day N= heading is treated as a 30-min
session.

Bindings:
- =C-c s n= → =rb/study-next= : from a file inside a plan
  directory, open the first TODO session in that plan; narrow,
  clock in, start a 30-min timer.
- =C-c s p= → =rb/study-next-in-plan= : prompt for which plan
  (subdirectory of =rb/study-plans-root=); default is the
  last-chosen plan. Same launch behaviour as =rb/study-next=.
- =C-c s d= → =rb/study-done= : mark current session DONE, clock
  out, stop timer, widen.

#+begin_src emacs-lisp
  (defcustom rb/study-plans-root "~/documents/org/study-plans"
    "Root directory under which study-plan subdirectories live.
  Each subdirectory is expected to contain `INDEX.org` plus
  `week-*.org` files."
    :type 'directory
    :group 'rb)

  (defvar rb/study-last-plan nil
    "Last study plan chosen via `rb/study-next-in-plan'.
  Used as the default for the next invocation.")

  (defun rb/study--plan-files (&optional dir)
    "Return week-*.org files in DIR, alphabetically.
  DIR defaults to the directory of the current buffer's file."
    (let ((d (or dir
                 (and buffer-file-name
                      (file-name-directory buffer-file-name)))))
      (when (and d (file-directory-p d))
        (sort (directory-files d t "^week-.*\\.org\\'") #'string<))))

  (defun rb/study--first-todo (&optional dir)
    "Return (FILE . POS) of next TODO heading across plan files in DIR.
  DIR defaults to the directory of the current buffer's file."
    (catch 'hit
      (dolist (f (rb/study--plan-files dir))
        (with-current-buffer (find-file-noselect f)
          (org-with-wide-buffer
           (goto-char (point-min))
           (when (re-search-forward
                  (concat "^\\*+ +" org-not-done-regexp) nil t)
             (throw 'hit (cons f (match-beginning 0)))))))
      nil))

  (defun rb/study--launch (hit)
    "Given (FILE . POS), open the file, narrow, clock in, start timer."
    (if hit
        (progn
          (find-file (car hit))
          (widen)
          (goto-char (cdr hit))
          (org-back-to-heading t)
          (org-show-entry)
          (org-narrow-to-subtree)
          (org-clock-in)
          (org-timer-set-timer "30"))
      (message "Plan complete — no TODO session left 🏁")))

  (defun rb/study-next ()
    "Open the next undone study session in the current plan.
  Run from any file inside a plan directory (typically the
  plan's INDEX.org).  Files matching =week-*.org= in the same
  directory are scanned in alphabetical order."
    (interactive)
    (rb/study--launch (rb/study--first-todo)))

  (defun rb/study--list-plans ()
    "Return list of plan-subdirectory names under `rb/study-plans-root'."
    (let ((root (expand-file-name rb/study-plans-root)))
      (when (file-directory-p root)
        (seq-filter
         (lambda (n)
           (and (not (string-prefix-p "." n))
                (file-directory-p (expand-file-name n root))))
         (directory-files root)))))

  (defun rb/study-next-in-plan ()
    "Prompt for a study plan, then open its next TODO session.
  Plans are subdirectories of `rb/study-plans-root'.  The chosen
  plan becomes the default for the next invocation, stored in
  `rb/study-last-plan'."
    (interactive)
    (let* ((plans (rb/study--list-plans)))
      (unless plans
        (user-error "No study plans found under %s" rb/study-plans-root))
      (let* ((default (or rb/study-last-plan (car plans)))
             (plan (completing-read
                    (format "Study plan (default %s): " default)
                    plans nil t nil nil default))
             (plan-dir (expand-file-name plan rb/study-plans-root)))
        (setq rb/study-last-plan plan)
        (rb/study--launch (rb/study--first-todo plan-dir)))))

  (defun rb/study-done ()
    "Mark the current session DONE, clock out, stop timer, widen."
    (interactive)
    (save-excursion
      (org-back-to-heading t)
      (org-todo "DONE"))
    (when (org-clocking-p) (org-clock-out))
    (when (and (boundp 'org-timer-current-timer) org-timer-current-timer)
      (org-timer-stop))
    (widen))

  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-c s n") #'rb/study-next)
    (define-key org-mode-map (kbd "C-c s p") #'rb/study-next-in-plan)
    (define-key org-mode-map (kbd "C-c s d") #'rb/study-done))
#+end_src

Open (or create on first visit) the weekly planning file for the
relevant ISO week. File path:
=<rb/weekly-planning-directory>/YYYY/Wnn.org=, where YYYY is the
ISO week-numbering year and =nn= is the zero-padded ISO week
number. The file gets a =#+TITLE= naming the Monday and Sunday
dates the first time it's created.

Behaviour: on Monday–Friday opens /this/ week; on Saturday or
Sunday opens /next/ week, so weekend planning runs on the week
about to start.

Binding:
- =C-c w= → =rb/plan-week=

#+begin_src emacs-lisp
  (defcustom rb/weekly-planning-directory "~/documents/org/weekly"
    "Root directory for weekly planning files.
  Files live at DIRECTORY/YYYY/Wnn.org where YYYY is the ISO
  week-numbering year and nn is the zero-padded ISO week number."
    :type 'directory
    :group 'rb)

  (defun rb/plan-week ()
    "Open the weekly planning file for the relevant week.
  Mon-Fri opens this week; Sat-Sun opens the upcoming week
  (so weekend planning targets the week about to start).

  The file is `rb/weekly-planning-directory'/YYYY/Wnn.org and is
  created with a `#+TITLE` heading on first visit."
    (interactive)
    (let* ((now (current-time))
           ;; ISO weekday: 1=Mon … 7=Sun.
           (dow (string-to-number (format-time-string "%u" now)))
           (offset (cond ((= dow 6) 2)        ; Sat → next Mon
                         ((= dow 7) 1)        ; Sun → next Mon
                         (t (- 1 dow))))      ; Mon-Fri → this Mon
           (mon (time-add now (* offset 86400)))
           (sun (time-add mon (* 6 86400)))
           (year (format-time-string "%G" mon))
           (week (format-time-string "%V" mon))
           (mon-str (format-time-string "%Y-%m-%d" mon))
           (sun-str (format-time-string "%Y-%m-%d" sun))
           (dir (expand-file-name year rb/weekly-planning-directory))
           (file (expand-file-name (format "w%s.org" week) dir))
           (fresh (not (file-exists-p file))))
      (make-directory dir t)
      (find-file file)
      (when fresh
        (insert (format "#+TITLE: Week %s: %s — %s\n\n" week mon-str sun-str))
        (save-buffer))))

  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-c w") #'rb/plan-week))
#+end_src

* Quick task entry
Add a TODO to the main agenda file with a back-link to the org section
(or file) currently being visited.  The destination heading inside the
tasks file is picked interactively via the standard ~org-refile~
machinery.

This is a plain ~org-capture~ template (key ~T~) built from two pieces:

- a ~file+function~ target, whose function runs the refile prompt and
  leaves point on the chosen heading — ~org-capture~ then files the
  entry as its last child (or as a top-level entry when the file itself
  is selected);
- a ~%(...)~ expansion that computes the back-link in the buffer the
  capture was started from, so ~org-id-get-create~ still stamps the
  right heading.

Going through ~org-capture~ means the usual capture affordances come for
free: ~C-c C-k~ to abort, editing the body/tags/scheduling before
finalising, ~C-u C-c C-w~ to jump back, and ~C-c c T~ as an alternative
entry point.

#+begin_src emacs-lisp
  (defcustom rb/tasks-file
    (expand-file-name "tasks.org" "~/documents/org/")
    "Agenda tasks file used by `rb/org-add-task'."
    :type 'file
    :group 'rb)

  (defun rb/org-add-task--back-link ()
    "Return an org link to the current heading (with ID) or the visited file."
    (cond
     ((and (derived-mode-p 'org-mode) (not (org-before-first-heading-p)))
      (let ((id (org-id-get-create))
            (title (org-get-heading t t t t)))
        (format "[[id:%s][%s]]" id title)))
     (buffer-file-name
      (let ((path (abbreviate-file-name buffer-file-name)))
        (format "[[file:%s][%s]]" path (file-name-nondirectory path))))
     (t
      (user-error "Buffer is not visiting a file and not under a heading"))))

  (defun rb/org-add-task--capture-link ()
    "Back-link to the buffer the current capture was initiated from.
Meant to be used from a `%(...)' expansion in a capture template."
    (let ((origin (or (org-capture-get :original-buffer) (current-buffer))))
      (if (buffer-live-p origin)
          (with-current-buffer origin (rb/org-add-task--back-link))
        (user-error "Capture origin buffer is gone"))))

  (defun rb/org-add-task--goto-target ()
    "Pick a heading in the current buffer with the `org-refile' machinery.
Leaves point on the chosen heading, so `org-capture' files the entry as
its child; when the file itself is selected, leaves point at the end of
the buffer on an empty line, which files the entry at top level."
    (let* ((org-refile-targets `((,(buffer-file-name (buffer-base-buffer))
                                  :maxlevel . 9)))
           (org-refile-use-outline-path 'file)
           (org-outline-path-complete-in-steps nil)
           (target (org-refile-get-location "Insert under"))
           (pos (nth 3 target)))
      (if pos
          (progn (goto-char pos) (org-back-to-heading t))
        (goto-char (point-max))
        (unless (bolp) (insert "\n")))))

  (defun rb/org-add-task ()
    "Capture a TODO into `rb/tasks-file' with a back-link to the current context.

The back-link targets the current org heading (creating an ID if needed)
when point is under one, or the visited file otherwise.  The destination
heading inside the tasks file is picked with the `org-refile' machinery."
    (interactive)
    (org-capture nil "T"))

  (with-eval-after-load 'org-capture
    (setq org-capture-templates
          (cons '("T" "Task (back-link to current context)" entry
                  (file+function rb/tasks-file rb/org-add-task--goto-target)
                  "* TODO %?\n%(rb/org-add-task--capture-link)")
                (seq-remove (lambda (entry) (equal (car entry) "T"))
                            org-capture-templates))))

  (with-eval-after-load 'org
    (define-key org-mode-map (kbd "C-c t") #'rb/org-add-task))
#+end_src
